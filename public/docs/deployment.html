3. Server Setup (Ubuntu 22.04)

Install PHP 8.4

sudo add-apt-repository ppa:ondrej/php -y
sudo apt update
sudo apt install php8.4 php8.4-fpm php8.4-cli php8.4-common \
    php8.4-mysql php8.4-zip php8.4-gd php8.4-mbstring \
    php8.4-curl php8.4-xml php8.4-bcmath php8.4-fileinfo \
    php8.4-tokenizer php8.4-dom -y

Install MySQL 8

sudo apt install mysql-server -y
sudo mysql_secure_installation

# Create database and user
sudo mysql -e "CREATE DATABASE bobby_holidays CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;"
sudo mysql -e "CREATE USER 'uniworld'@'localhost' IDENTIFIED BY 'YOUR_STRONG_PASSWORD';"
sudo mysql -e "GRANT ALL PRIVILEGES ON bobby_holidays.* TO 'uniworld'@'localhost';"
sudo mysql -e "FLUSH PRIVILEGES;"

Install Nginx

sudo apt install nginx -y

Install Composer

curl -sS https://getcomposer.org/installer | php
sudo mv composer.phar /usr/local/bin/composer

Install Supervisor (for queue worker)

sudo apt install supervisor -y

1Clone the repository 
cd /var/www
git clone YOUR_REPO_URL bobby-holidays
cd bobby-holidays
2Install PHP dependencies 
composer install --no-dev --optimize-autoloader
3Configure environment 
cp .env.example .env

Edit .env with production values:

5. Nginx Configuration

server {
    listen 80;
    server_name your-domain.com www.your-domain.com;
    root /var/www/bobby-holidays/public;
    index index.php;

    charset utf-8;
    client_max_body_size 20M;

    # Gzip compression
    gzip on;
    gzip_types text/plain text/css application/json application/javascript text/xml;
    gzip_min_length 1000;

    # Static file caching
    location ~* \.(jpg|jpeg|png|gif|ico|css|js|svg|woff2)$ {
        expires 30d;
        add_header Cache-Control "public, immutable";
    }

    # Block access to docs
    location /docs {
        auth_basic "Restricted";
        auth_basic_user_file /etc/nginx/.htpasswd;
    }

    location / {
        try_files $uri $uri/ /index.php?$query_string;
    }

    location ~ \.php$ {
        fastcgi_pass unix:/var/run/php/php8.4-fpm.sock;
        fastcgi_param SCRIPT_FILENAME $realpath_root$fastcgi_script_name;
        include fastcgi_params;
    }

    location ~ /\.(?!well-known).* {
        deny all;
    }
}

# Enable site and restart
sudo ln -s /etc/nginx/sites-available/bobby-holidays /etc/nginx/sites-enabled/
sudo nginx -t
sudo systemctl restart nginx

7. Cron Job (Scheduler)

sudo crontab -e

# Add this line:
* * * * * cd /var/www/bobby-holidays && php artisan schedule:run >> /dev/null 2>&1

This runs two scheduled commands:

  • quotations:expire — Daily at 00:30 (auto-expires past-validity quotations)
  • enquiries:follow-up-reminders — Daily at 09:00 (sends overdue follow-up notifications)

8. Queue Worker (Supervisor)

sudo nano /etc/supervisor/conf.d/bobby-holidays-worker.conf

[program:bobby-holidays-worker]
process_name=%(program_name)s_%(process_num)02d
command=php /var/www/bobby-holidays/artisan queue:work database --sleep=3 --tries=3 --max-time=3600
autostart=true
autorestart=true
stopasgroup=true
killasgroup=true
user=www-data
numprocs=1
redirect_stderr=true
stdout_logfile=/var/www/bobby-holidays/storage/logs/worker.log
stopwaitsecs=3600

sudo supervisorctl reread
sudo supervisorctl update
sudo supervisorctl start bobby-holidays-worker:*

13. Troubleshooting

500 Internal Server Error

# Check Laravel log
tail -50 storage/logs/laravel.log

# Check permissions
sudo chown -R www-data:www-data storage bootstrap/cache
sudo chmod -R 775 storage bootstrap/cache

# Clear all caches
php artisan optimize:clear

Styles/Assets Not Loading

# Check APP_URL in .env matches your domain
# Ensure storage link exists
php artisan storage:link

Email Not Sending

# Test mail config
php artisan tinker
> Mail::raw('Test', fn($m) => $m->to('test@example.com')->subject('Test'));

# Check MAIL_* settings in .env
# For Gmail: use App Password (not regular password)

Queue Jobs Not Processing

# Check supervisor status
sudo supervisorctl status

# Restart worker
sudo supervisorctl restart bobby-holidays-worker:*

# Process manually (debugging)
php artisan queue:work --once
